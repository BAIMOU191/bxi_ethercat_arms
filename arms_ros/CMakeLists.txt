cmake_minimum_required(VERSION 3.16)
project(arms_ros LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
foreach(pkg rclcpp sensor_msgs diagnostic_msgs communication)
  find_package(${pkg} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "srv/JointOperation.srv")
set(DRIVER ${CMAKE_CURRENT_SOURCE_DIR}/../arms_driver)
add_subdirectory(${DRIVER} driver EXCLUDE_FROM_ALL)
add_executable(arms_node src/arms_node.cpp ${DRIVER}/tools/arms_test/power.c ${DRIVER}/tools/arms_test/runtime_env.c ${DRIVER}/tools/arms_test/tlog.c)
target_include_directories(arms_node PRIVATE ${DRIVER}/tools/arms_test)
target_link_libraries(arms_node arms_driver)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$" AND EXISTS ${DRIVER}/third_party/bxi_pci_drv/arm64/libbxi_pci_drv.a)
  target_compile_definitions(arms_node PRIVATE ARMS_HAVE_PCI_POWER=1)
  target_include_directories(arms_node PRIVATE ${DRIVER}/third_party/bxi_pci_drv)
  target_link_libraries(arms_node ${DRIVER}/third_party/bxi_pci_drv/arm64/libbxi_pci_drv.a)
endif()
ament_target_dependencies(arms_node rclcpp communication sensor_msgs diagnostic_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(arms_node ${typesupport})
if(ETHERLAB_LIBRARY)
  get_filename_component(ETHERLAB_LIBDIR "${ETHERLAB_LIBRARY}" DIRECTORY)
  set_property(TARGET arms_node APPEND PROPERTY INSTALL_RPATH "${ETHERLAB_LIBDIR}")
endif()
install(TARGETS arms_node DESTINATION lib/${PROJECT_NAME})
install(FILES ${DRIVER}/config/arms.json DESTINATION share/${PROJECT_NAME}/config)
install(PROGRAMS ${DRIVER}/scripts/rt_setup.sh DESTINATION share/${PROJECT_NAME}/scripts)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
